For each of the equations below, consider which moves you _could_ make, then choose the best one and transform the equation. Repeat until you have solved for the unknown. _If you're not sure about an equation, try drawing them as Circles of Evaluation first!_

@n *@math{6d − 9 = −8}*

@indented{
Move: @fitb{12em}{@ifsoln{Additive Inverse}} @hspace{1em}
Transformed Equation: @fitb{}{@ifsoln{@math{6d − 9 + 9 = −8 + 9} &rarr; @math{6d = 1}}}

Move: @fitb{12em}{@ifsoln{Multiplicative Inverse}} @hspace{1em}
Transformed Equation: @fitb{}{@ifsoln{@math{\frac{1}{6} \times 6d = \frac{1}{6} \times 1} &rarr; @math{1d = \frac{1}{6}}}}

Move: @fitb{12em}{@ifsoln{Multiplicative Identity}} @hspace{1em}
Transformed Equation: @fitb{}{@ifsoln{@math{d = \frac{1}{6}}}}

Move: @fitb{12em}{} @hspace{1em}
Transformed Equation: @fitb{}{}

*Solve for @math{d}:* @fitb{}{@ifsoln{@math{d = \frac{1}{6}}}}
}

@n *@math{−11 = 2w − 7}*

@indented{
Move: @fitb{12em}{@ifsoln{Additive Inverse}} @hspace{1em}
Transformed Equation: @fitb{}{@ifsoln{@math{−11 + 7 = 2w − 7 + 7} &rarr; @math{−4 = 2w}}}

Move: @fitb{12em}{@ifsoln{Multiplicative Inverse}} @hspace{1em}
Transformed Equation: @fitb{}{@ifsoln{@math{−4 \times \frac{1}{2} = 2w \times \frac{1}{2}} &rarr; @math{−2 = 1w}}}

Move: @fitb{12em}{@ifsoln{Multiplicative Identity}} @hspace{1em}
Transformed Equation: @fitb{}{@ifsoln{@math{−2 = w}}}

Move: @fitb{12em}{} @hspace{1em}
Transformed Equation: @fitb{}{}

*Solve for @math{w}:* @fitb{}{@ifsoln{@math{w = −2}}}
}

@n *@math{\frac{r}{4} + 6 = 2}*

@indented{
Move: @fitb{12em}{@ifsoln{Additive Inverse}} @hspace{1em}
Transformed Equation: @fitb{}{@ifsoln{@math{\frac{r}{4} + 6 + (−6) = 2 + (−6)} &rarr; @math{\frac{r}{4} = −4}}}

Move: @fitb{12em}{@ifsoln{Multiplicative Inverse}} @hspace{1em}
Transformed Equation: @fitb{}{@ifsoln{@math{4 \times \frac{r}{4} = 4 \times (−4)} &rarr; @math{1r = −16}}}

Move: @fitb{12em}{@ifsoln{Multiplicative Identity}} @hspace{1em}
Transformed Equation: @fitb{}{@ifsoln{@math{r = −16}}}

Move: @fitb{12em}{} @hspace{1em}
Transformed Equation: @fitb{}{}

*Solve for @math{r}:* @fitb{}{@ifsoln{@math{r = −16}}}
}
